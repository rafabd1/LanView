cmake_minimum_required(VERSION 3.20)
project(lanview_clipboard LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
set(PROTOCOL ${CMAKE_CURRENT_SOURCE_DIR}/protocols/wlr-data-control-unstable-v1.xml)
add_custom_command(OUTPUT data-control-client.h COMMAND ${WAYLAND_SCANNER} client-header ${PROTOCOL} data-control-client.h DEPENDS ${PROTOCOL})
add_custom_command(OUTPUT data-control-protocol.c COMMAND ${WAYLAND_SCANNER} private-code ${PROTOCOL} data-control-protocol.c DEPENDS ${PROTOCOL})
add_executable(lanview-clipboard-wayland wayland.cpp ${CMAKE_CURRENT_BINARY_DIR}/data-control-protocol.c ${CMAKE_CURRENT_BINARY_DIR}/data-control-client.h)
target_include_directories(lanview-clipboard-wayland PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(lanview-clipboard-wayland PRIVATE PkgConfig::WAYLAND)
target_compile_options(lanview-clipboard-wayland PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS lanview-clipboard-wayland RUNTIME DESTINATION libexec/lanview)
install(PROGRAMS bridge.py DESTINATION libexec/lanview)
install(PROGRAMS lanview-clipboard DESTINATION bin RENAME lanview-bridge)
